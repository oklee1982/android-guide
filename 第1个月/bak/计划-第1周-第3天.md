fun String?.isValidName(): Boolean = !this.isNullOrBlank() && length in 1..20

3. 在按钮点击中执行校验与刷新：

```kotlin
binding.btnAdd.setOnClickListener {
    val name = binding.etName.text?.toString()?.trim()
    val ageInput = binding.etAge.text?.toString()?.trim()
    val age = ageInput.toValidAge()

    val error = when {
        !name.isValidName() -> "姓名需为 1-20 个字符"
        age == null -> "请输入 0-120 的年龄"
        users.any { it.name.equals(name, ignoreCase = true) } -> "姓名已存在"
        else -> null
    }

    if (error != null) {
        Toast.makeText(this, error, Toast.LENGTH_SHORT).show()
        return@setOnClickListener
    }

    users += User(name!!, age, city = null)
    binding.tvList.text = buildUserSummary(users)
    binding.etName.text?.clear()
    binding.etAge.text?.clear()
}
```
