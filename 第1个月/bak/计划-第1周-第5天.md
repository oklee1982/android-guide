binding.btnSave.setOnClickListener {
    val name = binding.etName.text?.toString()?.trim()
    val age = binding.etAge.text?.toString()?.trim()?.toIntOrNull()

    val error = when {
        !name.isValidName() -> "姓名需 1-20 个字符"
        age == null || age !in 0..120 -> "请输入合法年龄"
        else -> null
    }
    if (error != null) {
        showError(error)
        return@setOnClickListener
    }

    val user = User(name = name!!, age = age, city = null)
    val data = Intent().putExtra("extra_user", user)
    setResult(Activity.RESULT_OK, data)
    finish()
}

2. Repository 示例：

```kotlin
class UserRepository {
    private val _users = mutableListOf<User>()
    val users: List<User> get() = _users

    fun add(user: User): Boolean {
        if (_users.any { it.name.equals(user.name, ignoreCase = true) }) return false
        _users += user
        return true
    }

    fun remove(user: User): Boolean = _users.remove(user)
}
```
