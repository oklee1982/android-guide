- 使用集合操作（`map`, `filter`, `sortedBy`）生成展示文本，绑定到 TextView。示例：

```kotlin
private fun buildUserSummary(users: List<User>): String {
    return users
        .filter { it.age >= 0 }
        .sortedBy { it.name }
        .joinToString(separator = "\n") { user ->
            val cityInfo = user.city?.let { "（$it）" } ?: ""
            "${user.name} - ${user.age} 岁$cityInfo"
        }
}
```

实施步骤：

   - 在模块 `build.gradle.kts` 添加：

2. 修改 `User`：

```kotlin
import android.os.Parcelable
import kotlinx.parcelize.Parcelize

@Parcelize
data class User(
    val name: String,
    val age: Int,
    val city: String? = null
) : Parcelable
```

3. 创建 `SecondActivity`（布局可先用简单 TextView + Button）：

```kotlin
class SecondActivity : AppCompatActivity() {
    private lateinit var binding: ActivitySecondBinding

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        binding = ActivitySecondBinding.inflate(layoutInflater)
        setContentView(binding.root)

        val user = intent.getParcelableExtra<User>("extra_user")
        if (user == null) {
            Toast.makeText(this, "用户信息缺失", Toast.LENGTH_SHORT).show()
            finish()
            return
        }
        binding.tvUserDetail.text = """
            姓名：${user.name}
            年龄：${user.age}
            城市：${user.city ?: "未填写"}
        """.trimIndent()
    }
}
```
